function parseMultiRecipeDocument(text) { const lines = text.split(/\r?\n/).map(l => l.trim()); const blocks = []; let currentBlock = []; for (let i = 0; i < lines.length; i++) { const line = lines[i]; if (!line) { if (currentBlock.length > 0) currentBlock.push(''); // preserve empty line for paragraph separation continue; } const isSeparator = /^([=\-_*#]{3,})$/.test(line); const looksLikeRecipeHeader = /^Recipe\s*[:\d]/i.test(line) || /^#{1,3}\s+.*recipe/i.test(line) || (line.toUpperCase() === line && line.length > 5 && line.length < 65 && !line.includes(':')); if ((isSeparator || (looksLikeRecipeHeader && currentBlock.length > 8)) && currentBlock.length > 0) { blocks.push(currentBlock); currentBlock = []; if (!isSeparator) currentBlock.push(line); } else { currentBlock.push(line); } } if (currentBlock.length > 0) blocks.push(currentBlock); return blocks.map(block => parseSingleRecipeBlock(block)).filter(Boolean); } function parseSingleRecipeBlock(lines) { if (lines.length < 3) return null; // Filter out leading empty lines while (lines.length > 0 && !lines[0].trim()) lines.shift(); let title = lines[0] .replace(/^#+\s*/, '') .replace(/^RECIPE\s*[:.]\s*/i, '') .replace(/\*+/g, '') .trim(); let mode = 'preamble'; const ingredients = []; const rawInstructionParagraphs = []; let currentStepBuffer = []; let prepTime = ''; let cookTime = ''; let servings = ''; for (let i = 1; i < lines.length; i++) { const rawLine = lines[i]; const line = rawLine.trim(); if (!line) { // Empty line signals step boundary if we're in instruction mode if (mode === 'instructions' && currentStepBuffer.length > 0) { rawInstructionParagraphs.push(currentStepBuffer.join(' ')); currentStepBuffer = []; } continue; } const cleanLower = line.toLowerCase().replace(/[*_#]/g, '').trim(); // Section triggers if (/^(ingredients|provisions|what you('ll| will) need)[:.]?$/i.test(cleanLower)) { if (mode === 'instructions' && currentStepBuffer.length > 0) { rawInstructionParagraphs.push(currentStepBuffer.join(' ')); currentStepBuffer = []; } mode = 'ingredients'; continue; } else if (/^(instructions|directions|method|preparation|steps)[:.]?$/i.test(cleanLower)) { mode = 'instructions'; continue; } // Metadata extraction if (cleanLower.startsWith('prep time:')) { prepTime = line.split(':')[1]?.replace(/[*_]/g, '').trim() || ''; continue; } if (cleanLower.startsWith('cook time:')) { cookTime = line.split(':')[1]?.replace(/[*_]/g, '').trim() || ''; continue; } if (cleanLower.startsWith('servings:') || cleanLower.startsWith('yield:')) { servings = line.split(':')[1]?.replace(/[*_]/g, '').trim() || ''; continue; } if (mode === 'ingredients') { // Ignore subheadings like "### Overnight Pizza Crust Ingredients" or "### Sauce" if (/^#+\s+/.test(line) && !line.match(/^#+\s+\d/)) continue; if (line.endsWith(':') && !line.match(/\d/)) continue; const ingObj = parseIngredientLine(line); if (ingObj) ingredients.push(ingObj); } else if (mode === 'instructions') { // Check if this line is the beginning of a brand-new numbered or bulleted step const isNewNumberedStep = /^(?:step\s*)?\d+[\.\)]\s+/i.test(line); const isBulletedSubstep = /^[-*•]\s+/.test(line); if (isNewNumberedStep) { if (currentStepBuffer.length > 0) { rawInstructionParagraphs.push(currentStepBuffer.join(' ')); currentStepBuffer = []; } // Strip the leading number marker for a clean step const cleanContent = line.replace(/^(?:step\s*)?\d+[\.\)]\s+/i, '').replace(/[*_]/g, '').trim(); currentStepBuffer.push(cleanContent); } else if (isBulletedSubstep) { // Sub-bullets (like "- Slow Cooker Method: ...") are preserved with clear indentation const cleanBullet = line.replace(/^[-*•]\s+/, '').replace(/[*_]/g, '').trim(); if (currentStepBuffer.length > 0) { currentStepBuffer.push('— ' + cleanBullet); } else { currentStepBuffer.push(cleanBullet); } } else { // Continuation of current step const cleanText = line.replace(/[*_]/g, '').trim(); if (currentStepBuffer.length > 0) { currentStepBuffer.push(cleanText); } else if (cleanText.length > 15) { // Unnumbered instruction line currentStepBuffer.push(cleanText); } } } } // Flush remaining buffer if (currentStepBuffer.length > 0) { rawInstructionParagraphs.push(currentStepBuffer.join(' ')); } if (ingredients.length === 0) return null; return { id: 'rec_import_' + Math.random().toString(36).substring(2, 9), title: title || 'Untitled Imported Dish', prepTime: prepTime || '15 mins', cookTime: cookTime || '25 mins', servings: servings || '4 servings', tags: ['Dinner', 'Gluten-Free', 'Imported'], photo: '', ratings: { taste: 4, ease: 4, presentation: 4, leftovers: 4 }, tweaks: '', reviews: [], ingredients, instructions: rawInstructionParagraphs.length > 0 ? rawInstructionParagraphs : ['Follow standard cooking and assembly instructions.'], createdAt: Date.now() }; }
